def require(ok,message):
    if not ok:raise RuntimeError(message)
def digest(raw):return hashlib.sha256(raw).hexdigest()
def canonical(value):return (json.dumps(value,sort_keys=True,indent=2,allow_nan=False)+'\n').encode()
def write(path,raw):path.parent.mkdir(parents=True,exist_ok=True);path.write_bytes(raw)
def utc():return dt.datetime.now(dt.timezone.utc)
def load(name,path,expected):
    raw=path.read_bytes();require(digest(raw)==expected,'control module source hash differs')
    module=ModuleType(name);module.__file__=str(path)
    exec(compile(raw,str(path),'exec',dont_inherit=True,optimize=sys.flags.optimize),module.__dict__);return module

def invoke(ctx,lane,scope,call_directory,case,action,mode=None,expected_hash=None,target='campaign.py'):
    require(utc()<ctx['deadline'],'operational deadline exhausted before instrumented entrypoint')
    call_directory.mkdir(parents=True,exist_ok=False)
    if target=='campaign.py':
        source=(lane/'candidate/Candidate.lean').read_bytes() if expected_hash is None else None
        argv=[action,'--candidate-sha256',expected_hash or digest(source)]
    else:argv=['--lane',str(lane)]
    job={'lane':str(lane),'scope':str(scope),'target':str(lane/target),'case':case,'argv':argv,'marker_sha256':digest((lane/'SYNTHETIC_CONTROL_ONLY.json').read_bytes())}
    write(call_directory/'JOB.json',canonical(job))
    flags=sys.flags.optimize if mode is None else mode
    command=[sys.executable,'-I','-S','-B']+(['-O'] if flags else [])+[str(LANE/'instrumented-child.py'),str(call_directory/'JOB.json')]
    runner=ctx['runtime'].Runner(call_directory,{'PATH':'/usr/bin:/bin','LANG':'C.UTF-8','LC_ALL':'C.UTF-8'},preparation=False)
    try:runner.run(command,call_directory,'instrumented-entrypoint',timeout=min(120,(ctx['deadline']-utc()).total_seconds()),allow_stdout=True)
    except ctx['runtime'].JudgeError:pass
    require(len(runner.commands)==1,'instrumented supervisor child missing')
    child=runner.commands[0];require(child['failure'] is None and child['remaining_live_group_processes']==[],'instrumented supervisor operational failure')
    require((call_directory/'DRIVER_RESULT.json').is_file(),'instrumentation driver receipt missing')
    driver=json.loads((call_directory/'DRIVER_RESULT.json').read_bytes())
    require(driver.get('status')=='instrumented_entrypoint_completed' and type(driver.get('external_process_attempts')) is list
            and driver['external_process_attempts']==[] and type(driver.get('new_lean_children')) is int and driver['new_lean_children']==0
            and type(driver.get('new_proofs')) is int and driver['new_proofs']==0,'instrumentation attempted external execution or has incomplete typed metadata')
    expected_status=next((status for name,_,status,_ in CASES if name==case),None) if target=='campaign.py' else None
    require(target!='campaign.py' or expected_status is not None,'unregistered instrumented campaign case')
    expected_code=(1 if expected_status=='failed' else 0) if target=='campaign.py' else (0 if case=='positive-baseline' else 1)
    require(type(driver.get('returncode')) is int and type(child.get('returncode')) is int
            and driver['returncode']==child['returncode']==expected_code,'instrumented terminal code differs')
    results=sorted((lane/'attempts').glob('*/RESULT.json')) if (lane/'attempts').exists() else []
    result=json.loads(results[-1].read_bytes()) if results else None
    if result is None:
        early=sorted((lane/'entrypoints').glob('*/RESULT.json'));result=json.loads(early[-1].read_bytes()) if early else None
    if target=='campaign.py':require(type(result) is dict and result.get('status')==expected_status and result.get('dispatch')==action,'instrumented intended result status differs')
    return driver,result

def create_matrix(ctx,directory):
    lane,source=fixture(ctx,directory,True);calls=directory/'calls'

    invoke(ctx,lane,directory,calls/'00-failed','compiler-operational-failure','development')

    for mode in (0,1):invoke(ctx,lane,directory,calls/('04-cosmetic-'+str(mode)),'full-positive','full',mode)
    return lane

def finish(ctx,output,status,cases):
    require(utc()<ctx['deadline'],'operational deadline exhausted before control custody')
    artifacts={str(path.relative_to(output)):{'bytes':path.stat().st_size,'sha256':digest(path.read_bytes())} for path in sorted(output.rglob('*')) if path.is_file()}
    require(utc()<ctx['deadline'],'operational deadline exhausted during control custody')
    for name,raw in ctx['original'].items():require((LANE/name).read_bytes()==raw,'control package drift during execution')
    result={'status':status,'python_optimization':sys.flags.optimize,'package_sha256':digest(ctx['package_raw']),
            'self_sha256':digest(Path(sys.argv[0]).read_bytes()),'case_count':len(cases),'cases':cases,'new_lean_children':0,'new_proofs':0,
            'artifacts':artifacts,'completed_utc':utc().isoformat(),'scope':'Whole source-bound instrumented entrypoints with inert synthetic candidates/objects; external execution denied inside driver; zero proof credit.'}
    write(output/'RESULT.json',canonical(result));print(canonical({'status':status,'case_count':len(cases),'new_proofs':0}).decode(),end='')
