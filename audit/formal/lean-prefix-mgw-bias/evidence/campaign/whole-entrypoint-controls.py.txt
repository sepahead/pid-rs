def digest(raw): return hashlib.sha256(raw).hexdigest()
def canon(value): return (json.dumps(value, sort_keys=True, indent=2, allow_nan=False) + '\n').encode()
def require(ok, message):

def write(path, raw): path.parent.mkdir(parents=True, exist_ok=True); path.write_bytes(raw)
def now(): return datetime.datetime.now(datetime.timezone.utc)
def load_module(name, path, expected):
    raw = path.read_bytes()
    if hashlib.sha256(raw).hexdigest() != expected: raise RuntimeError('module bytes differ: ' + str(path))
    module = ModuleType(name); module.__file__ = str(path)
    exec(compile(raw, str(path), 'exec', dont_inherit=True, optimize=sys.flags.optimize), module.__dict__)
    return module
