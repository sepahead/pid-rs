"""Root-only strict closure of the exact five-family bias campaign; no tool execution."""
from __future__ import annotations
import argparse, datetime, fcntl, hashlib, json, math, re, stat, sys
from pathlib import Path
from types import ModuleType

def require(ok, message):
    if not ok: raise RuntimeError(message)
def digest(raw): return hashlib.sha256(raw).hexdigest()
def canonical(value): return (json.dumps(value, sort_keys=True, indent=2, allow_nan=False) + '\n').encode()
def strict(raw):
    def pairs(xs):
        result = {}
        for key, value in xs:
            require(key not in result, 'duplicate JSON key'); result[key] = value
        return result
    return json.loads(raw, object_pairs_hook=pairs, parse_constant=lambda value: (_ for _ in ()).throw(ValueError('nonfinite JSON: '+value)))
def same(a, b):
    if type(a) is not type(b): return False
    if type(a) is dict: return set(a) == set(b) and all(same(a[k], b[k]) for k in a)
    if type(a) is list: return len(a) == len(b) and all(same(x, y) for x, y in zip(a, b))
    return a == b

def obj(value, fields, label): require(type(value) is dict and set(value) == set(fields), label + ' fields differ')
def integer(value, label, low=None):
    require(type(value) is int and (low is None or value >= low), label + ' must be an exact integer')
def number(value, label, low=0):
    require(type(value) in (int, float) and math.isfinite(value) and value >= low, label + ' must be finite numeric')
def string(value, label): require(type(value) is str and bool(value), label + ' must be a nonempty string')
def hashed(value, label): require(type(value) is str and re.fullmatch('[0-9a-f]{64}', value) is not None, label + ' must be a SHA256 string')
def stamp(value, label):
    string(value, label); result = datetime.datetime.fromisoformat(value)
    require(result.tzinfo is not None, label + ' must be timezone aware'); return result

def main():
    require(sys.flags.isolated and sys.flags.no_site and sys.dont_write_bytecode and sys.flags.optimize in (0, 1), 'requires -I -S -B optionally -O')
    parser = argparse.ArgumentParser(description=__doc__); parser.add_argument('--lane', required=True, type=Path); args = parser.parse_args()
    lane = args.lane.absolute(); captured = {}
    require(not (lane / 'SYNTHETIC_CONTROL_ONLY.json').exists(), 'synthetic control lane cannot run live')
    lock = (lane / 'EXECUTION.lock').open('a+'); fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
    def read(path):
        require(path == path.resolve(strict=True) and stat.S_ISREG(path.lstat().st_mode) and path.stat().st_nlink == 1, 'noncanonical evidence input')
        raw = path.read_bytes(); previous = captured.setdefault(str(path), raw); require(previous == raw, 'input drift during closure'); return raw
    def within(base, name):
        string(name, 'retained path'); path = Path(name)
        require(not path.is_absolute() and '..' not in path.parts and str(path) == name, 'unsafe retained relative path'); return base / path
    def load(name, filename, sha):
        path = lane / filename; raw = read(path); require(digest(raw) == sha, 'policy source drift')
        module = ModuleType(name); module.__file__ = str(path)
        exec(compile(raw, str(path), 'exec', dont_inherit=True, optimize=sys.flags.optimize), module.__dict__); return module
    require(not (lane / 'STOP_RECORD.json').exists() and not (lane / 'ACCEPTANCE.json').exists(), 'campaign already terminal')
    task_raw = read(lane / 'CAMPAIGN.json'); task = strict(task_raw)
    obj(task, ['schema','registered_utc','deadline_utc','project_root','freeze_sha256','max_development_attempts','max_full_attempts','max_concurrent_wrappers','candidate_write_scope'], 'campaign')
    require(task['schema'] == 'pid-rs/prefix-mgw-bias-campaign-v1' and task['candidate_write_scope'] == 'candidate/Candidate.lean', 'campaign identity differs')
    for field, expected in [('max_development_attempts',80),('max_full_attempts',8),('max_concurrent_wrappers',1)]:
        integer(task[field], 'campaign.'+field); require(task[field] == expected, 'fixed campaign bound differs')
    started = stamp(task['registered_utc'],'campaign start'); deadline = stamp(task['deadline_utc'],'campaign deadline')
    now = lambda: datetime.datetime.now(datetime.timezone.utc)
    require(deadline-started == datetime.timedelta(minutes=720) and started <= now() < deadline, 'campaign clock invalid or exhausted')
    frozen_raw = read(lane / 'FROZEN.json'); frozen = strict(frozen_raw)
    obj(frozen,['schema','status','candidate_exists','wrapper_sha256'],'freeze')
    hashed(task['freeze_sha256'],'campaign.freeze_sha256')
    require(task['freeze_sha256'] == digest(frozen_raw) and frozen['schema'] == 'pid-rs/prefix-mgw-bias-runner-freeze-v1'
            and frozen['status'] == 'adopted_after_actual_controls_before_candidate' and frozen['candidate_exists'] is False,
            'live freeze differs')
    require(type(frozen['wrapper_sha256']) is dict, 'freeze pins must be an object')
    for name, expected in frozen['wrapper_sha256'].items():
        hashed(expected,'freeze digest'); require(digest(read(within(lane,name))) == expected, 'adopted source or prerequisite drift')
    graph = strict(read(lane / 'SOURCE_GRAPH.json')); modules = graph['dependencies'] + graph['trusted_types']
    require(type(modules) is list and len(modules) == 27 and len(set(modules)) == 27, 'source graph differs')
    for field, expected in [('full_children',31),('wrong_final_children',29),('development_children',1)]:
        integer(graph[field],'graph.'+field); require(graph[field] == expected, 'graph bound differs')
    observed = strict(read(lane / 'HISTORICAL_ENVIRONMENT.json'))
    lean = next(name for name in observed['tools'] if name.endswith('/bin/lean'))
    checker = next(name for name in observed['tools'] if name.endswith('/bin/leanchecker'))
    pins = frozen['wrapper_sha256']
    policies = {name:load('closure_'+name, filename, pins[filename]) for name,filename in
                [('bias','policy.py'),('mean','mean-policy.py'),('prefix','prefix-policy.py'),('mgw','mgw-policy.py')]}
    accepted = strict(read(lane / 'ACCEPTED_DEPENDENCY_RECORDS.json'))
    ledger = strict(read(lane / 'ATTEMPT_LEDGER.json')); obj(ledger,['campaign_sha256','attempts'],'ledger')
    require(type(ledger['attempts']) is list and ledger['campaign_sha256'] == digest(task_raw), 'ledger identity differs')
    full, wrong, examined = [], [], []
    for index, entry in enumerate(ledger['attempts']):
        obj(entry,['registered_utc','kind','slot_kind','mode','candidate_sha256','output','prospective_compiler_kernel_children'],'attempt registration')
        require(type(entry['kind']) is str and entry['kind'] in ('development','full','wrong-last-target'), 'attempt kind differs')
        integer(entry['mode'],'attempt.mode'); require(entry['mode'] in (0,1), 'attempt mode differs')
        expected_slot = 'development' if entry['kind'] == 'development' else 'full'
        require(entry['slot_kind'] == expected_slot and entry['output'] == 'attempts/'+str(index+1).zfill(3)+'-'+entry['kind']+'-'+str(entry['mode']), 'attempt path or slot differs')
        integer(entry['prospective_compiler_kernel_children'],'attempt.prospective_compiler_kernel_children')
        count = 1 if entry['kind'] == 'development' else 29 if entry['kind'] == 'wrong-last-target' else 31
        require(entry['prospective_compiler_kernel_children'] == count, 'attempt graph bound differs')
        attempt_start = stamp(entry['registered_utc'],'attempt start'); require(started <= attempt_start < deadline, 'attempt clock differs')
        hashed(entry['candidate_sha256'],'attempt candidate hash')
        directory = within(lane,entry['output']); result_raw = read(directory / 'RESULT.json'); result = strict(result_raw)
        mandatory={'status','dispatch','new_bias_proofs','registered','commands','inputs','artifacts','completed_utc'}
        optional={'captured_sha256','executed_sha256','prior_full_pass','semantic_records','accepted_mgw_dependency_records',
                  'accepted_prefix_dependency_records','accepted_mean_dependency_records','preserved_semantic_records',
                  'failure','final_custody_failure','artifact_failure'}
        require(type(result) is dict and mandatory <= set(result) <= mandatory | optional, 'attempt result fields differ')
        for field in ('failure','final_custody_failure','artifact_failure'):
            if field in result: require(type(result[field]) is str, 'failure metadata type differs')
        require(type(result) is dict and type(result.get('status')) is str and result['status'] in
                ('failed','development_compile_pass','fresh_semantic_kernel_pass','expected_target_rejection'), 'attempt status differs')
        require(same(result.get('registered'),entry) and same(strict(read(directory/'REGISTERED.json')),entry), 'attempt registration correspondence differs')
        integer(result.get('new_bias_proofs'),'result.new_bias_proofs'); require(result['new_bias_proofs'] == 0, 'per-attempt proof credit differs')
        require(result.get('dispatch') == entry['kind'], 'attempt dispatch differs')
        completed = stamp(result.get('completed_utc'),'attempt completion'); require(attempt_start <= completed, 'attempt completion predates start')
        if result['status'] != 'failed': require(completed < deadline, 'accepted attempt completed after deadline')
        # All attempts, including failed and diagnostic development, retain complete evidence.
        require(type(result.get('artifacts')) is dict and type(result.get('inputs')) is dict, 'attempt custody maps differ')
        actual_names=set()
        for path in directory.rglob('*'):
            mode=path.lstat().st_mode
            require(stat.S_ISDIR(mode) or stat.S_ISREG(mode), 'nonregular retained artifact')
            if stat.S_ISREG(mode):actual_names.add(str(path.relative_to(directory)))
        # Only this attempt's top-level RESULT is excluded.
        actual_names.discard('RESULT.json')
        require(set(result['artifacts']) == actual_names, 'retained artifact roster differs')
        for name, record in result['artifacts'].items():
            obj(record,['bytes','sha256'],'artifact record'); integer(record['bytes'],'artifact.bytes',0); hashed(record['sha256'],'artifact.sha256')
            body = read(within(directory,name)); require(len(body) == record['bytes'] and digest(body) == record['sha256'], 'retained artifact drift')
        for name, record in result['inputs'].items():
            string(name,'input source path'); require(Path(name).is_absolute(),'input source path must be absolute'); obj(record,['bytes','sha256','identity'],'input record')
            integer(record['bytes'],'input.bytes',0); hashed(record['sha256'],'input.sha256')
            require(type(record['identity']) is list and len(record['identity']) == 7, 'input identity differs')
            for value in record['identity']: integer(value,'input.identity',0)
            require(record['identity'][3]==1 and record['identity'][4]==record['bytes'] and stat.S_ISREG(record['identity'][2]),'saved input identity shape differs')
            body = read(directory/'input-bytes'/record['sha256'])
            require(len(body) == record['bytes'] and digest(body) == record['sha256'], 'retained source preimage drift')
        prior_ledger = strict(read(directory/'LEDGER.before.json'))
        require(same(prior_ledger,{'campaign_sha256':digest(task_raw),'attempts':ledger['attempts'][:index]}), 'saved prior ledger differs')
        source_root=directory/'src'
        for module in modules:
            path=source_root/(module+'.lean')
            if path.exists():require(read(path)==read(lane/('sources/'+module+'.lean')),'compiled dependency source differs from freeze')
        for module,template in [(graph['semantic_judge'],'SemanticJudge.lean.in'),(graph['root'],'BiasCandidateRoot.lean.in')]:
            path=source_root/(module+'.lean')
            if path.exists():require(read(path)==read(lane/template),'compiled trusted judge/root source differs from freeze')
        commands = result.get('commands'); require(type(commands) is list and len(commands) <= count, 'command count differs')
        sequence = ([] if entry['kind'] == 'development' else modules) + [graph['candidate']]
        if entry['kind'] != 'development': sequence += [graph['semantic_judge']]
        if entry['kind'] == 'full': sequence += [graph['root'], None]
        streams = {}
        for child_index, command in enumerate(commands):
            fields=['command','cwd','label','started_utc','elapsed_seconds','returncode','failure','execution_error','limits','peak_observed_group_rss_bytes','remaining_live_group_processes','stdout_sha256','stderr_sha256','retained_stdout_bytes','retained_stderr_bytes']
            obj(command,fields,'command summary')
            integer(command['returncode'],'command.returncode') if command['returncode'] is not None else require(result['status']=='failed','accepted command missing returncode')
            for field in ('retained_stdout_bytes','retained_stderr_bytes','peak_observed_group_rss_bytes'): integer(command[field],'command.'+field,0)
            number(command['elapsed_seconds'],'command.elapsed_seconds'); child_start=stamp(command['started_utc'],'command start')
            require(attempt_start <= child_start <= completed, 'command time outside attempt')
            for field in ('failure','execution_error'): require(command[field] is None or type(command[field]) is str, 'command.'+field+' type differs')
            require(type(command['remaining_live_group_processes']) is list, 'process roster type differs')
            for process in command['remaining_live_group_processes']:
                require(type(process) is list and len(process)==3, 'process record differs'); integer(process[0],'process.pid',1);integer(process[1],'process.rss',0);string(process[2],'process.state')
            limits=command['limits'];obj(limits,['seconds','observed_group_rss_bytes','rss_sample_interval_seconds','rss_limit_is_hard_os_cap','stdout_bytes','stderr_bytes'],'command limits')
            number(limits['seconds'],'limits.seconds');require(0<limits['seconds']<=600,'child timeout differs')
            for field,expected in [('observed_group_rss_bytes',4*1024**3),('stdout_bytes',4*1024**2),('stderr_bytes',4*1024**2)]:
                integer(limits[field],'limits.'+field);require(limits[field]==expected,'child resource limit differs')
            require(type(limits['rss_sample_interval_seconds']) is float and limits['rss_sample_interval_seconds']==0.1 and limits['rss_limit_is_hard_os_cap'] is False,'RSS policy differs')
            module=sequence[child_index];src=directory/'src';build=directory/'build'
            label='fresh-kernel' if module is None else 'compile-root' if module==graph['root'] else 'compile-'+module.replace('/','-')
            argv=[checker,'--fresh',graph['root']] if module is None else [lean,'-t','0','-M','4096','-o',str(build/(module+'.olean')),str(src/(module+'.lean'))]
            require(same(command['command'],argv) and command['cwd']==str(src) and command['label']==label,'command graph/argv/cwd differs')
            child=directory/(str(child_index).zfill(2)+'-'+label)
            retained=strict(read(child/'command.json'));require(same(command,retained),'command summary differs from retained command.json')
            registration=strict(read(child/'registered.json'))
            obj(registration,['registered_utc','command','cwd','output','preparation','source_inputs'],'child registration')
            require(same(registration['command'],argv) and registration['cwd']==str(src) and registration['output']==str(child) and registration['preparation'] is False,'child registration binding differs')
            require(attempt_start<=stamp(registration['registered_utc'],'child registration time')<=completed,'child registration time differs')
            expected_sources={str(path.relative_to(src)):digest(read(path)) for path in sorted(src.rglob('*.lean'))}
            require(set(expected_sources)=={module+'.lean' for module in modules+[graph['candidate'],graph['semantic_judge'],graph['root']]},'child complete source roster differs')
            require(same(registration['source_inputs'],expected_sources),'child source-input binding differs')
            for stream in ('stdout','stderr'):
                body=read(child/(stream+'.log'));hashed(command[stream+'_sha256'],'command stream hash')
                require(digest(body)==command[stream+'_sha256'] and len(body)==command['retained_'+stream+'_bytes'],'command stream binding differs')
                require(len(body)<=limits[stream+'_bytes'],'retained stream exceeds bound')
                if stream=='stdout':streams[label]=body
        examined.append({'output':entry['output'],'status':result['status'],'result_sha256':digest(result_raw)})
        if result['status'] == 'failed': continue
        require(len(commands)==count,'successful attempt graph incomplete')
        require('failure' not in result and 'final_custody_failure' not in result and 'artifact_failure' not in result,'success carries failure evidence')
        positive=commands[:-1] if entry['kind']=='wrong-last-target' else commands
        for command in positive:
            if '-o' in command['command']: read(Path(command['command'][command['command'].index('-o')+1]))
            require(command['returncode']==0 and command['failure'] is None and command['execution_error'] is None
                    and command['remaining_live_group_processes']==[] and command['retained_stderr_bytes']==0,'positive child did not finish cleanly')
        original=read(directory/'Candidate.original.lean');executed=read(directory/'Candidate.executed.lean')
        for field in ('captured_sha256','executed_sha256'):hashed(result.get(field),'result.'+field)
        require(digest(original)==result['captured_sha256']==entry['candidate_sha256'] and digest(executed)==result['executed_sha256'],'candidate byte correspondence differs')
        execution=strict(read(directory/'EXECUTION_INPUT.json'));require(same(execution,{'registered':entry,'executed_sha256':digest(executed)}),'execution input binding differs')
        require(read(directory/'src'/(graph['candidate']+'.lean'))==executed,'executed candidate source differs')
        if result['status']=='development_compile_pass':
            require(entry['kind']=='development' and original==executed,'development identity differs');continue
        for family,judge in [('mgw','PidMgwBridge'),('prefix','PidPrefixProbability'),('mean','PidPrefixMgwMean')]:
            parsed=policies[family].semantic_records(streams['compile-'+judge+'-SemanticJudge'],strict)
            require(same(parsed,accepted[family]) and same(parsed,result.get('accepted_'+family+'_dependency_records')),'inherited record receipt binding differs')
        item={'entry':entry,'result':result,'result_sha256':digest(result_raw),'original':original,'executed':executed,'directory':directory}
        semantic=streams['compile-PidPrefixMgwBias-SemanticJudge']
        if result['status']=='fresh_semantic_kernel_pass':
            require(entry['kind']=='full' and original==executed,'full pass identity differs')
            parsed=policies['bias'].semantic_records(semantic,strict);require(same(parsed,result.get('semantic_records')),'bias semantic stream differs');full.append(item)
        else:
            require(entry['kind']=='wrong-last-target','negative kind differs');last=commands[-1]
            require(last['returncode']==1 and last['failure'] is None and last['execution_error'] is None and last['retained_stderr_bytes']==0 and last['remaining_live_group_processes']==[],'negative control ended operationally')
            text=semantic.decode();lines=[x for x in text.splitlines() if x.startswith('PREFIX_MGW_BIAS_JUDGE_RESULT ')]
            require(len(lines)==4 and 'mass_floor_bias' in text and 'type mismatch' in text.lower(),'wrong-final causal stream differs')
            records=[strict(x.split(' ',1)[1].encode()) for x in lines];require(same(records,result.get('preserved_semantic_records')),'wrong-final preserved stream differs')
            prior=result.get('prior_full_pass');obj(prior,['path','sha256'],'prior full receipt');hashed(prior['sha256'],'prior receipt hash')
            matches=[x for x in full if str(x['directory']/'RESULT.json')==prior['path'] and x['result_sha256']==prior['sha256'] and x['original']==original]
            require(matches and same(records,matches[0]['result']['semantic_records'][:4]),'wrong-final prior causal receipt differs');wrong.append(item)
    require(sum(x['slot_kind']=='development' for x in ledger['attempts'])<=80 and sum(x['slot_kind']=='full' for x in ledger['attempts'])<=8,'ledger attempt budget exceeded')
    require(full,'no complete selected candidate');selected=full[0]['original'];selected_records=full[0]['result']['semantic_records']
    cosmetic=b'/- Statement-preserving formatting control. -/\n\n'+selected+b'\n';matrix=[]
    for label,source in [('selected',selected),('cosmetic',cosmetic)]:
        for mode in (0,1):
            matches=[x for x in full if x['original']==source and x['entry']['mode']==mode]
            require(matches,'missing fresh-kernel matrix cell: '+label+':'+str(mode));item=matches[0]
            require(same(item['result']['semantic_records'],selected_records),'matrix semantic records differ')
            matrix.append({'source':label,'mode':mode,'output':item['entry']['output'],'result_sha256':item['result_sha256']})
    controls=[x for x in wrong if x['original']==selected and same(x['result']['preserved_semantic_records'],selected_records[:4])]
    require(controls,'actual wrong-final four-record preservation absent');negative=controls[0]
    ending=b'theorem mass_floor_bias : True := by trivial\n\nend PidPrefixMgwBiasCandidate\n'
    require(negative['executed'].endswith(ending) and selected.startswith(negative['executed'][:-len(ending)]),'negative source is not trusted final replacement')
    for name,raw in list(captured.items()):require(Path(name).read_bytes()==raw,'evidence drift during closure')
    require(now()<deadline and not (lane/'STOP_RECORD.json').exists(),'campaign expired or stopped before closure')
    acceptance={'status':'accepted_complete_five_bias_families','selected_sha256':digest(selected),'cosmetic_sha256':digest(cosmetic),'semantic_records':selected_records,
                'matrix':matrix,'wrong_final':{'output':negative['entry']['output'],'result_sha256':negative['result_sha256']},'examined_attempts':examined,
                'new_bias_families':5,'new_dependency_families':0,'extra_proofs_from_repeat_runs_or_controls':0,'completed_utc':now().isoformat(),
                'inputs':{name:{'bytes':len(raw),'sha256':digest(raw)} for name,raw in sorted(captured.items())}}
    raw=canonical(acceptance)
    with (lane/'ACCEPTANCE.json').open('xb') as stream:stream.write(raw)
    with (lane/'STOP_RECORD.json').open('xb') as stream:stream.write(canonical({'status':'complete','acceptance_sha256':digest(raw),'completed_utc':now().isoformat(),'scope':'five frozen families only'}))
    print(canonical({'status':acceptance['status'],'acceptance_sha256':digest(raw)}).decode(),end='')
if __name__=='__main__':main()
