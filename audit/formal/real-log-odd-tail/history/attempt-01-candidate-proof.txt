  have h_lower (y : ℝ) (h_a : 0 < y) (h_b : y < 1) : 0 < halfTail n y := by
    have h_next := Real.sum_range_le_log_div h_a.le h_b (n + 1)
    rw [Finset.sum_range_succ] at h_next
    have h_term : 0 < y ^ (2 * n + 1) / (2 * (n : ℝ) + 1) := by positivity
    dsimp [halfTail]
    linarith
  by_cases h_zero : x = 0
  · rw [h_zero]
    simp [halfTail, upper]
  · have h_pos : 0 < x := lt_of_le_of_ne h₀ h_zero.symm
    let g (t : ℝ) := t ^ (2 * n + 1) / (2 * (n : ℝ) + 1) - (1 - t ^ 2) * halfTail n t
    have h_square (y : ℝ) (h_a : 0 ≤ y) (h_b : y < 1) : 0 < 1 - y ^ 2 := by
      have h_minus : 0 < 1 - y := by linarith
      have h_plus : 0 < 1 + y := by linarith
      nlinarith [mul_pos h_minus h_plus]
    have h_coefficient : 0 < 2 * (n : ℝ) + 1 := by positivity
    have h_deriv (y : ℝ) (h_y : y ∈ Icc 0 x) : HasDerivAt g (2 * y * halfTail n y) y := by
      have h_right : y < 1 := lt_of_le_of_lt h_y.2 h₁
      have h_left : -1 < y := by linarith [h_y.1]
      have h_den : 1 - y ^ 2 ≠ 0 := by linarith [h_square y h_y.1 h_right]
      have h_coef : 2 * (n : ℝ) + 1 ≠ 0 := by linarith
      have h_tail : HasDerivAt (halfTail n) ((y ^ 2) ^ n / (1 - y ^ 2)) y :=
        Real.hasDerivAt_half_log_one_add_div_one_sub_sub_sum_range n h_left h_right
      have h_power : HasDerivAt (fun t : ℝ => t ^ (2 * n + 1) / (2 * (n : ℝ) + 1))
          (y ^ (2 * n)) y := by
        apply ((hasDerivAt_pow (2 * n + 1) y).div_const (2 * (n : ℝ) + 1)).congr_deriv
        have h_exp : 2 * n + 1 - 1 = 2 * n := by omega
        simp [h_exp, Nat.cast_add, Nat.cast_mul] <;> field_simp [h_coef] <;> ring
      apply (h_power.sub (((hasDerivAt_pow 2 y).const_sub (1 : ℝ)).mul h_tail)).congr_deriv
      simp [pow_mul] <;> field_simp [h_den] <;> ring
    have h_strong : StrictMonoOn g (Icc 0 x) := by
      apply strictMonoOn_of_hasDerivWithinAt_pos (convex_Icc 0 x)
        (fun y h_y => (h_deriv y h_y).continuousAt.continuousWithinAt)
        (fun y h_y => (h_deriv y (interior_subset h_y)).hasDerivWithinAt)
      intro y h_y
      simp only [interior_Icc, Set.mem_Ioo] at h_y
      exact mul_pos (mul_pos (by norm_num) h_y.1) (h_lower y h_y.1 (lt_trans h_y.2 h₁))
    have h_cmp : g 0 < g x := h_strong ⟨le_rfl, h₀⟩ ⟨h₀, le_rfl⟩ h_pos
    have h_atzero : g 0 = 0 := by simp [g, halfTail]
    rw [h_atzero] at h_cmp
    change 0 < x ^ (2 * n + 1) / (2 * (n : ℝ) + 1) - (1 - x ^ 2) * halfTail n x at h_cmp
    have h_fin : (1 - x ^ 2) * halfTail n x < x ^ (2 * n + 1) / (2 * (n : ℝ) + 1) := by linarith
    have h_upper : halfTail n x < upper n x := by
      apply (lt_div_iff₀ (mul_pos h_coefficient (h_square x h₀ h₁))).2
      calc
        halfTail n x * ((2 * (n : ℝ) + 1) * (1 - x ^ 2)) =
            ((1 - x ^ 2) * halfTail n x) * (2 * (n : ℝ) + 1) := by ring
        _ < x ^ (2 * n + 1) := (lt_div_iff₀ h_coefficient).1 h_fin
    exact ⟨(h_lower x h_pos h₁).le, h_upper.le, fun h_input => ⟨h_lower x h_input h₁, h_upper⟩⟩
