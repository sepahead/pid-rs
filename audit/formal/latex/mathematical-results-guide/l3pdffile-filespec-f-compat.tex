% This Work adapts the two file-specification writers in l3pdffile.dtx from
% latex3/pdfresources commit f1a1b29c4e355ee0786f67ca364c123ff6aa622e,
% "l3pdffiles: use utf8/string in F key for better compability with zugferd"
% (2024-02-24), licensed under LPPL 1.3c or later. Commit
% 57caeb67fa785a92a05083f4eee1863d504f7a63 subsequently identifies the first
% release containing this change as v0.96f dated 2024-03-01. See
% LICENSE-LPPL-1.3c.
%
% Ubuntu's reviewed pdfmanagement-testphase.ltx v0.96d dated 2023-12-09 writes
% both /F and /UF as UTF-16BE PDF text strings. The upstream change instead
% writes /F through utf8/string and retains the print-string conversion for
% /UF. For this guide's ASCII relative PDF targets, that yields canonical
% ASCII /F bytes while preserving the UTF-16BE /UF value, decoded target, and
% GoToR destination. This compatibility layer does not alter URI or internal
% GoTo actions, link labels, structure parents, or the guide's PDF/UA boundary.
%
% The native path wins. We inspect both writer bodies, reject mixed states,
% reject native behavior on an unexpectedly older file, and reject a native-
% absent file at or after the first reviewed release. The fallback accepts only
% the exact date/version token family 2023-12-09 v0.96d, whose hosted source
% is separately bound by SHA-256. It requires every upstream API before
% installing the two source-faithful writer definitions and refreshing the
% public alias made earlier by the bundle.
\ExplSyntaxOn
% A reviewed native bundle must not allocate fallback state or change the
% file-specification writers. Probe it without allocating a message, sentinel,
% variable, register, or helper control sequence, then stop reading this Work
% before fallback state is created. Reading an extra input can change pdfTeX's
% automatic trailer /ID; the self-test therefore requires exact equality
% outside the sole typed, duplicated trailer-ID payload and separately checks
% action semantics. Missing, mixed, unexpectedly old, or unrecognized shapes
% fall through to the diagnostic and exact-old branch below.
\cs_if_exist:cT { ver@pdfmanagement-testphase.ltx }
  {
    \cs_if_exist:NT \__pdffile_filespec_write:nnn
      {
        \cs_if_exist:NT \__pdffile_filespec_write:nnN
          {
            \cs_if_exist:NT \pdffile_filespec:nnn
              {
                \cs_if_eq:NNT
                  \pdffile_filespec:nnn
                  \__pdffile_filespec_write:nnn
                  {
                    \exp_args:Ne \tl_if_in:nnT
                      { \cs_meaning:N \__pdffile_filespec_write:nnn }
                      { pdf_string_from_unicode:nnN }
                      {
                        \exp_args:Ne \tl_if_in:nnT
                          { \cs_meaning:N \__pdffile_filespec_write:nnn }
                          { utf8/string }
                          {
                            \exp_args:Ne \tl_if_in:nnT
                              { \cs_meaning:N \__pdffile_filespec_write:nnN }
                              { pdf_string_from_unicode:nnN }
                              {
                                \exp_args:Ne \tl_if_in:nnT
                                  { \cs_meaning:N \__pdffile_filespec_write:nnN }
                                  { utf8/string }
                                  {
                                    \IfFileAtLeastTF
                                      { pdfmanagement-testphase.ltx }
                                      { 2024-03-01 }
                                      { \ExplSyntaxOff \endinput }
                                      { }
                                  }
                              }
                          }
                      }
                  }
              }
          }
      }
  }
