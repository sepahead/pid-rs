\msg_new:nnn { pid-rs } { tagpdf-openaction-compat }
  { Mathematical~results~guide~tagpdf~compatibility~failed:~#1. }
\cs_new_protected:Npn \__pidrs_tagpdf_openaction_fail:n #1
  { \msg_fatal:nnn { pid-rs } { tagpdf-openaction-compat } {#1} }

\cs_if_exist:NTF \g__pidrs_tagpdf_openaction_compat_loaded_bool
  { \__pidrs_tagpdf_openaction_fail:n { duplicate~compatibility~application } }
  { \bool_new:N \g__pidrs_tagpdf_openaction_compat_loaded_bool }
\bool_gset_true:N \g__pidrs_tagpdf_openaction_compat_loaded_bool

\cs_if_exist:NF \__tag_tree_update_openaction:
  {
    \cs_if_exist:cTF { ver@tagpdf.sty }
      { }
      { \__pidrs_tagpdf_openaction_fail:n { tagpdf~is~not~loaded } }
    % LaTeX's "later" test is inclusive, so 2024/02/23 is the first rejected
    % package date and the 2024/02/22 v0.98w release remains eligible.
    \use:c { @ifpackagelater } { tagpdf } { 2024/02/23 }
      {
        \__pidrs_tagpdf_openaction_fail:n
          { newer~tagpdf~lacks~the~reviewed~native~OpenAction~update }
      }
      { }
    \cs_if_exist:NF \g__tag_active_tree_bool
      { \__pidrs_tagpdf_openaction_fail:n { tagpdf~active-tree~state~is~absent } }
    \cs_if_exist:NF \__kernel_pdfdict_name:n
      { \__pidrs_tagpdf_openaction_fail:n { PDF~catalog~lookup~API~is~absent } }
    \cs_if_exist:NF \pdfmanagement_add:nne
      { \__pidrs_tagpdf_openaction_fail:n { PDF~catalog~update~API~is~absent } }
    \cs_if_exist:NF \pdf_object_ref:n
      { \__pidrs_tagpdf_openaction_fail:n { PDF~object-reference~API~is~absent } }

    \tl_new:N \l__pidrs_tagpdf_openaction_tl
    \seq_new:N \l__pidrs_tagpdf_openaction_seq
    \cs_new_protected:Npn \__pidrs_tagpdf_update_openaction:
      {
        \prop_get:cnNT
          { \__kernel_pdfdict_name:n { g__pdf_Core/Catalog } }
          { OpenAction }
          \l__pidrs_tagpdf_openaction_tl
          {
            \tl_if_head_eq_charcode:eNT
              { \tl_trim_spaces:V \l__pidrs_tagpdf_openaction_tl } [ % ]
              {
                \seq_set_split:NnV
                  \l__pidrs_tagpdf_openaction_seq
                  { / }
                  \l__pidrs_tagpdf_openaction_tl
                \pdfmanagement_add:nne { Catalog } { OpenAction }
                  {
                    <<
                      /S/GoTo \c_space_tl
                      /D~\l__pidrs_tagpdf_openaction_tl \c_space_tl
                      /SD~[\pdf_object_ref:n { __tag/struct/1 }
                        \int_compare:nNnTF
                          { \seq_count:N \l__pidrs_tagpdf_openaction_seq } > { 1 }
                          { /\seq_item:Nn \l__pidrs_tagpdf_openaction_seq { 2 } }
                          { ] }
                    >>
                  }
              }
          }
      }
    \hook_gput_code:nnn { shipout/lastpage } { tagpdf }
      {
        \bool_if:NT \g__tag_active_tree_bool
          { \__pidrs_tagpdf_update_openaction: }
      }
  }
