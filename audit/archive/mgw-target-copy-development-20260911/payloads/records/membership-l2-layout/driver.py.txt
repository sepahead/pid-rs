def need(ok, why):
    if not ok:
        raise RuntimeError(why)


def same_json(actual, expected, why):
    need(json.dumps(actual, sort_keys=True, allow_nan=False) ==
         json.dumps(expected, sort_keys=True, allow_nan=False), why)


def load_capture():
    before = CAPTURE.lstat()
    need(CAPTURE.resolve(strict=True) == CAPTURE and stat.S_ISREG(before.st_mode)
         and before.st_size == 15326, "capture source path/size")
    with CAPTURE.open("rb") as handle:
        raw = handle.read(15327)
    need(len(raw) == 15326 and hashlib.sha256(raw).hexdigest() == CAPTURE_SHA,
         "capture source bytes")
    module = types.ModuleType("pid_mgw_target_copy_qualification_capture")
    module.__file__ = str(CAPTURE)
    exec(compile(raw, str(CAPTURE), "exec"), module.__dict__)
    return module


def output_observation(capture, row, raw):
    need(b"\0" not in raw and b"\r" not in raw, "binary/non-LF compiler stdout")
    if row["id"] != "P05":
        need(raw == b"", "candidate emitted stdout")
        return {"empty_stdout": True}
    prefix = b"MGW_TARGET_COPY_JUDGE "
    need(raw.startswith(prefix) and raw.endswith(b"\n") and raw.count(b"\n") == 1,
         "not one complete judge record")
    report = capture.parse(raw[len(prefix):-1])
    need(type(report) is dict and set(report) == {"schema", "status", "count", "theorems"},
         "judge report shape")
    need(report["schema"] == "pid-rs-mgw-target-copy-l1-l4-judge-report-v1" and
         report["status"] == "type-and-axiom-checks-passed-pending-actual-outcome-and-root-review" and
         type(report["count"]) is int and report["count"] == 4 and
         type(report["theorems"]) is list and len(report["theorems"]) == 4, "judge roster")
    keys = {"candidate_fqn", "target_fqn", "candidate_kind", "proof_value_present",
            "candidate_universes", "target_universes", "candidate_full_type",
            "expected_closed_target", "original_target_type", "original_target_value",
            "full_type_defeq", "candidate_axioms", "target_axioms"}
    for value, (candidate, target) in zip(report["theorems"], BINDINGS, strict=True):
        need(type(value) is dict and set(value) == keys and
             value["candidate_fqn"] == candidate and value["target_fqn"] == target and
             value["candidate_kind"] == "theorem" and value["proof_value_present"] is True and
             value["full_type_defeq"] is True, "binding/kind/type/proof result")
        for key in ("candidate_universes", "target_universes"):
            levels = value[key]
            need(type(levels) is list and len(levels) == 2 and
                 all(type(level) is str and level for level in levels) and len(set(levels)) == 2,
                 "two distinct universe parameters")
        for key in ("candidate_axioms", "target_axioms"):
            axioms = value[key]
            need(type(axioms) is list and all(type(a) is str for a in axioms) and
                 axioms == sorted(set(axioms)) and
                 set(axioms) <= {"propext", "Classical.choice", "Quot.sound"}, "axiom closure")
        for key in ("candidate_full_type", "expected_closed_target", "original_target_type",
                    "original_target_value"):
            need(type(value[key]) is str and value[key].strip() and
                 "⋯" not in value[key] and "..." not in value[key], "missing/elided declaration")
    return {"report": report, "root_semantic_review": "pending"}


def current_objects(capture, manifest, stage):
    seeded = {str(stage / name) for name in (
        "objects/PidMgwTargetCopy/Interface.olean",
        "objects/PidMgwTargetCopyJudge/JudgeCore.olean")}
    pins = {str(stage / pin["stage_relative"]): pin for pin in manifest["stage_inputs"]
            if pin["stage_relative"].startswith("objects/")}
    need(set(pins) == seeded, "exact two seeded import objects")
    paths = capture.members(stage / "objects")
    need(paths is not None and seeded <= paths and len(paths - seeded) <= 5,
         "output membership bound")
    for path, pin in pins.items():
        capture.observe(path, dict(pin, path=path))
    result = {}
    for path in sorted(paths - seeded):
        record, _ = capture.observe(path, cap=2 * 1024**3)
        need(record["bytes"] > 0, "empty output object")
        result[path] = record
    need(sum(pin["bytes"] for pin in result.values()) <= 2 * 1024**3,
         "new-object byte bound")
    return result


def selected_inputs(capture, manifest, phase, state):
    pins = manifest["fixed_inputs"] + manifest["stage_inputs"]
    need(len(pins) == 23 and len({pin["path"] for pin in pins}) == 23,
         "selected-input roster")
    state["selected_inputs"][phase] = {"status": "running", "checked": []}
    for pin in pins:
        record, _ = capture.observe(pin["path"], pin)
        state["selected_inputs"][phase]["checked"].append(record)
    state["selected_inputs"][phase]["status"] = "observed-selected-inputs-only"
