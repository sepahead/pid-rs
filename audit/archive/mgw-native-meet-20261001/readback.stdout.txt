MgwNativeMeetProofs.native_source_membership.{u, v, w} {ι : Type u} {V : ι → Type v} {T : Type w} [Fintype ι]
  [(i : ι) → Fintype (V i)] [Fintype T] [DecidableEq ι] [(i : ι) → DecidableEq (V i)] [DecidableEq T]
  (F : MgwMeetContract.Family ι) (anchor state : PidFiniteConvergence.CategoricalKey ι V T) :
  state ∈ PidFiniteConvergence.sxSourceEvent F anchor ↔
    MgwMeetContract.SourceEvent F (fun i key => key.1 i) anchor state
MgwNativeMeetProofs.native_source_meet_union.{u, v, w} {ι : Type u} {V : ι → Type v} {T : Type w} [Fintype ι]
  [(i : ι) → Fintype (V i)] [Fintype T] [DecidableEq ι] [(i : ι) → DecidableEq (V i)] [DecidableEq T]
  (A B : MgwMeetContract.Family ι) (anchor : PidFiniteConvergence.CategoricalKey ι V T) :
  PidFiniteConvergence.sxSourceEvent (MgwMeetContract.Minimal (A ∪ B)) anchor =
    PidFiniteConvergence.sxSourceEvent A anchor ∪ PidFiniteConvergence.sxSourceEvent B anchor
MgwNativeMeetProofs.native_target_restricted_meet_union.{u, v, w} {ι : Type u} {V : ι → Type v} {T : Type w} [Fintype ι]

  PidFiniteConvergence.sxTargetRestrictedEvent (MgwMeetContract.Minimal (A ∪ B)) anchor =
    PidFiniteConvergence.sxTargetRestrictedEvent A anchor ∪ PidFiniteConvergence.sxTargetRestrictedEvent B anchor

MgwNativeMeetControls.source_not_target_witness :
  have F := {{0}};
  have anchor := (fun x => false, false);
  have state := (fun x => false, true);
  state ∈ PidFiniteConvergence.sxSourceEvent F anchor ∧ state ∉ PidFiniteConvergence.sxTargetRestrictedEvent F anchor
MgwNativeMeetControls.same_anchor_witness :
  have F := {{0}};
  have anchor := (fun x => false, false);
  have otherSource := (fun x => true, false);
  have otherTarget := (fun x => false, true);
  (otherSource ∉ PidFiniteConvergence.sxSourceEvent (MgwMeetContract.Minimal (F ∪ F)) anchor ∧
      otherSource ∈ PidFiniteConvergence.sxSourceEvent F anchor ∪ PidFiniteConvergence.sxSourceEvent F otherSource) ∧
    otherTarget ∉ PidFiniteConvergence.sxTargetRestrictedEvent (MgwMeetContract.Minimal (F ∪ F)) anchor ∧
      otherTarget ∈
        PidFiniteConvergence.sxTargetRestrictedEvent F anchor ∪
          PidFiniteConvergence.sxTargetRestrictedEvent F otherTarget
