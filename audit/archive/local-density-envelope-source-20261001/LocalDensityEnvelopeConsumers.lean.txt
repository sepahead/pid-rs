theorem mean_bounds :
  ∀ (α : Type u) [MeasurableSpace α] (μ : Measure α) (E : Set α)
    (f : α → ℝ) (a ε : ℝ),
    MeasurableSet E →
    IsFiniteMeasure (μ.restrict E) →
    0 < μ.real E →
    IntegrableOn f E μ →
    0 < a →
    0 ≤ ε →
    (∀ᵐ x ∂(μ.restrict E),
      a * Real.exp (-ε) ≤ f x ∧ f x ≤ a * Real.exp ε) →
    a * Real.exp (-ε) ≤ LocalDensityEnvelopeContractDesign.localMean μ E f ∧
      LocalDensityEnvelopeContractDesign.localMean μ E f ≤ a * Real.exp ε
  := @LocalDensityEnvelopeProofs.mean_bounds

  IsFiniteMeasure (LocalDensityEnvelopeContractDesign.controlMeasure.restrict Set.univ) ∧
  LocalDensityEnvelopeContractDesign.controlMeasure.real Set.univ = 1 ∧
  IntegrableOn (fun _ : ℝ => (1 : ℝ)) Set.univ LocalDensityEnvelopeContractDesign.controlMeasure ∧
  (∀ᵐ _x ∂(LocalDensityEnvelopeContractDesign.controlMeasure.restrict Set.univ),
    (1 : ℝ) * Real.exp 0 ≤ 1 ∧ 1 ≤ (1 : ℝ) * Real.exp 0) ∧
  (∫ x in Set.univ, (fun _ : ℝ => (1 : ℝ)) x ∂LocalDensityEnvelopeContractDesign.controlMeasure) = 1 ∧
  LocalDensityEnvelopeContractDesign.localMean LocalDensityEnvelopeContractDesign.controlMeasure Set.univ (fun _ : ℝ => (1 : ℝ)) = 1 ∧
  |Real.log (LocalDensityEnvelopeContractDesign.localMean LocalDensityEnvelopeContractDesign.controlMeasure Set.univ (fun _ : ℝ => (1 : ℝ)) / 1)| = 0
  := @LocalDensityEnvelopeProofs.positive_constant_density
