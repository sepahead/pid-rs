#!/usr/bin/env bash
set -euo pipefail

SCRIPT_DIRECTORY="${BASH_SOURCE[0]%/*}"
if [[ "$SCRIPT_DIRECTORY" == "${BASH_SOURCE[0]}" ]]; then
  SCRIPT_DIRECTORY="."
fi
ROOT="$(cd "$SCRIPT_DIRECTORY/.." && pwd -P)"
CHECKER="$ROOT/scripts/check-formal-pdf-log.sh"
TMP_ROOT="${TMPDIR:-/tmp}"
FIXTURE_ROOT="$(mktemp -d "$TMP_ROOT/pid-rs-formal-pdf-log-self-test.XXXXXX")"
trap 'rm -rf -- "$FIXTURE_ROOT"' EXIT
ACCEPTED_COUNT=0
REJECTED_COUNT=0

require_accept() {
  local name="$1"
  local line="$2"
  local log="$FIXTURE_ROOT/$name.log"
  local stdout="$FIXTURE_ROOT/$name.stdout"
  local stderr="$FIXTURE_ROOT/$name.stderr"
  printf '%s\n' "$line" >"$log"
  if ! bash "$CHECKER" "$log" >"$stdout" 2>"$stderr"; then
    echo "formal PDF log self-test: accepted control failed: $name" >&2
    return 1
  fi
  if [[ -s "$stdout" || -s "$stderr" ]]; then
    echo "formal PDF log self-test: accepted control emitted output: $name" >&2
    return 1
  fi
  ACCEPTED_COUNT=$((ACCEPTED_COUNT + 1))
}

require_reject() {
  local name="$1"
  local line="$2"
  local log="$FIXTURE_ROOT/$name.log"
  local stdout="$FIXTURE_ROOT/$name.stdout"
  local stderr="$FIXTURE_ROOT/$name.stderr"
  local expected="$FIXTURE_ROOT/$name.expected"
  local terminated="${3:-true}"
  local status
  if [[ "$terminated" == "false" ]]; then
    printf '%s' "$line" >"$log"
  else
    printf '%s\n' "$line" >"$log"
  fi
  printf '%s\n' "$line" >"$expected"
  set +e
  bash "$CHECKER" "$log" >"$stdout" 2>"$stderr"
  status=$?
  set -e
  if [[ "$status" -ne 1 ]]; then
    echo "formal PDF log self-test: rejected mutation returned $status: $name" >&2
    return 1
  fi
  if [[ -s "$stdout" ]] || ! cmp -s "$expected" "$stderr"; then
    echo "formal PDF log self-test: rejected mutation output drifted: $name" >&2
    return 1
  fi
  REJECTED_COUNT=$((REJECTED_COUNT + 1))
}

require_intermediate_accept() {
  local name="$1"
  local line="$2"
  local log="$FIXTURE_ROOT/$name.log"
  local stdout="$FIXTURE_ROOT/$name.stdout"
  local stderr="$FIXTURE_ROOT/$name.stderr"
  printf '%s\n' "$line" >"$log"
  if ! bash "$CHECKER" --intermediate "$log" >"$stdout" 2>"$stderr"; then
    echo "formal PDF log self-test: intermediate convergence control failed: $name" >&2
    return 1
  fi
  if [[ -s "$stdout" || -s "$stderr" ]]; then
    echo "formal PDF log self-test: intermediate convergence control emitted output: $name" >&2
    return 1
  fi
  ACCEPTED_COUNT=$((ACCEPTED_COUNT + 1))
}

require_intermediate_reject() {
  local name="$1"
  local line="$2"
  local log="$FIXTURE_ROOT/$name.log"
  local stdout="$FIXTURE_ROOT/$name.stdout"
  local stderr="$FIXTURE_ROOT/$name.stderr"
  local expected="$FIXTURE_ROOT/$name.expected"
  local terminated="${3:-true}"
  local expected_line="${4:-$line}"
  local status
  if [[ "$terminated" == "false" ]]; then
    printf '%s' "$line" >"$log"
  else
    printf '%s\n' "$line" >"$log"
  fi
  printf '%s\n' "$expected_line" >"$expected"
  set +e
  bash "$CHECKER" --intermediate "$log" >"$stdout" 2>"$stderr"
  status=$?
  set -e
  if [[ "$status" -ne 1 || -s "$stdout" ]] || ! cmp -s "$expected" "$stderr"; then
    echo "formal PDF log self-test: intermediate mutation did not fail closed: $name" >&2
    return 1
  fi
  REJECTED_COUNT=$((REJECTED_COUNT + 1))
}

require_accept "clean-output" "Output written on paper.pdf (12 pages, 123456 bytes)."
require_accept "informational-message" "Package microtype Info: Character ignored."
require_accept "font-information" "LaTeX Font Info: Font shape declared on input line 7."
require_accept "engine-information" "LuaTeX Info: deterministic object stream enabled."

require_reject "latex-warning" "LaTeX Warning: Reference undefined on input line 7."
require_reject "package-warning" "Package hyperref Warning: Token not allowed in a PDF string."
require_reject "class-warning" "Class article Warning: Unsupported class option on input line 1."
require_reject "font-warning" "LaTeX Font Warning: Font shape unavailable on input line 9."
require_reject "pdftex-warning" "pdfTeX warning (dest): name{section.1} has been referenced but does not exist"
require_reject "luatex-warning" "LuaTeX warning (file missing.pdf): PDF inclusion: required page does not exist"
require_reject "xdvipdfmx-warning" "xdvipdfmx:warning: Unparsed material at end of special ignored."
require_reject "missing-character" "Missing character: There is no U+2212 in font LatinModernRoman!"
require_reject "overfull-hbox" 'Overfull \hbox (2.0pt too wide) in paragraph at lines 1--2'
require_reject "underfull-hbox" 'Underfull \hbox (badness 10000) in paragraph at lines 1--2'
require_reject "overfull-vbox" 'Overfull \vbox (33.7033pt too high) has occurred while \output is active []'
require_reject "underfull-vbox" 'Underfull \vbox (badness 10000) has occurred while \output is active []'
require_reject "undefined-references" "There were undefined references."
require_reject "undefined-citations" "There were undefined citations."
require_reject "multiply-defined" "There were multiply defined labels."
require_reject "rerun-outlines" "Rerun to get outlines right"
require_reject "latex-error" "! LaTeX Error: File missing.sty not found."
require_reject "package-error" "! Package graphicx Error: Unknown graphics extension."
require_reject "emergency-stop" "! Emergency stop."
require_reject "runaway-argument" "Runaway argument?"
require_reject "undefined-control" "! Undefined control sequence."
require_reject "fatal-error" "Fatal error occurred, no output PDF file produced!"

require_intermediate_accept \
  "intermediate-reference" \
  "LaTeX Warning: Reference \`sec:one' on page 1 undefined on input line 7."
require_intermediate_accept \
  "intermediate-undefined-summary" \
  "LaTeX Warning: There were undefined references."
require_intermediate_accept \
  "intermediate-cross-reference-rerun" \
  "LaTeX Warning: Label(s) may have changed. Rerun to get cross-references right."
require_intermediate_accept \
  "intermediate-outline-file" \
  "Package rerunfilecheck Warning: File \`paper.out' has changed."
require_intermediate_accept \
  "intermediate-outline-rerun" \
  "(rerunfilecheck)                Rerun to get outlines right"
require_intermediate_reject \
  "intermediate-package-warning" \
  "Package hyperref Warning: Token not allowed in a PDF string."
require_intermediate_reject \
  "intermediate-overfull" \
  'Overfull \hbox (2.0pt too wide) in paragraph at lines 1--2'


require_both_accept() {
  require_accept "$1" "$2"
  require_intermediate_accept "$1" "$2"
}

require_both_reject() {
  require_reject "$1" "$2" "${3:-true}"
  require_intermediate_reject "$1" "$2" "${3:-true}"
}

# Actual stock information is not a diagnostic merely because it says warning.
require_both_accept "stock-infwarerr" \
  "Package: infwarerr 2019/12/03 v1.5 Providing info/warning/error messages (HO)"
require_both_accept "stock-error-style-information" \
  "file:line:error style messages enabled."
require_both_accept "warning-macro-information" \
  'LaTeX Info: Redefining \PackageWarning on input line 20.'
require_both_accept "warning-counter" '\c@warningcount=\count100'
require_both_accept "warning-prose" \
  "Ordinary informational note about warning handling."
require_both_accept "warning-filename" "(/tmp/warning-support.sty"

# Preserve the exact bare V3 header shape, including its trailing space.
require_both_reject "bare-pdf-inclusion" \
  "warning  (file order-implication.pdf) (pdf inclusion): PDF inclusion: found PDF "
require_both_reject "bare-warning-colon" "warning: unexpected diagnostic"
require_both_reject "uppercase-bare-warning" "WARNING: unexpected diagnostic"
require_both_reject "mixed-case-bare-warning" \
  "wArNiNg (pdf backend): unexpected diagnostic"
require_both_reject "indented-bare-warning" $'\t  warning: unexpected diagnostic'
require_both_reject "standalone-bare-warning" "warning"
require_both_reject "bare-warning-crlf" $'warning\r'
require_both_reject "bare-warning-no-final-newline" "warning: unexpected" false
require_both_reject "generic-missing-number" "! Missing number, treated as zero."
require_both_reject "generic-dimension" "! Dimension too large."
require_both_reject "generic-extra-brace" "! Extra }."
require_both_reject "indented-bang-error" $'\t! Improper alphabetic constant.'
require_both_reject "standalone-bang-error" "!"
require_both_reject "bang-error-crlf" $'! Missing number, treated as zero.\r'
require_both_reject "bang-error-no-final-newline" "! Dimension too large." false

# The five named convergence records are exceptions only in intermediate mode.
# ANSI-C quoting makes the quote delimiters literal; no command substitution.
CONVERGENCE_RECORDS=(
  $'LaTeX Warning: Reference \x60sec:one\x27 on page 1 undefined on input line 7.'
  'LaTeX Warning: There were undefined references.'
  'LaTeX Warning: Label(s) may have changed. Rerun to get cross-references right.'
  $'Package rerunfilecheck Warning: File \x60paper.out\x27 has changed.'
  '(rerunfilecheck)                Rerun to get outlines right'
)
for index in "${!CONVERGENCE_RECORDS[@]}"; do
  record="${CONVERGENCE_RECORDS[$index]}"
  require_reject "strict-convergence-$index" "$record"
  require_intermediate_accept "crlf-convergence-$index" "$record"$'\r'
  require_intermediate_reject "mixed-prefix-$index" "warning: hidden; $record"
  require_intermediate_reject "mixed-suffix-$index" "$record warning: hidden"
done
require_both_reject "reference-field-injection" \
  $'LaTeX Warning: Reference \x60sec:one warning: hidden\x27 on page 1 undefined on input line 7.'
require_both_reject "reference-record-injection" \
  $'LaTeX Warning: Reference \x60sec:one\x27 Package x Warning: hidden undefined on input line 7.'
require_both_reject "file-field-injection" \
  $'Package rerunfilecheck Warning: File \x60paper.out warning: hidden\x27 has changed.'
require_both_reject "file-error-suffix" \
  $'Package rerunfilecheck Warning: File \x60paper.out\x27 has changed. ! Missing number, treated as zero.'
require_both_reject "rerun-continuation-injection" \
  "(rerunfilecheck) warning: hidden Rerun to get outlines right"
require_both_reject "empty-reference-label" \
  $'LaTeX Warning: Reference \x60\x27 on page 1 undefined on input line 7.'
require_both_reject "invalid-reference-page" \
  $'LaTeX Warning: Reference \x60sec:one\x27 on page x undefined on input line 7.'
require_both_reject "empty-rerun-file" \
  $'Package rerunfilecheck Warning: File \x60\x27 has changed.'
require_both_reject "unregistered-rerun-suffix" \
  "(rerunfilecheck)                Rerun to get outlines right unknown suffix"
MIXED_LINES=$'LaTeX Warning: Label(s) may have changed. Rerun to get cross-references right.\nwarning: hidden'
require_reject "allowed-plus-newline-warning" "$MIXED_LINES"
require_intermediate_reject "allowed-plus-newline-warning" "$MIXED_LINES" true \
  "warning: hidden"


set +e
bash "$CHECKER" "$FIXTURE_ROOT/absent.log" >"$FIXTURE_ROOT/missing.stdout" \
  2>"$FIXTURE_ROOT/missing.stderr"
MISSING_STATUS=$?
bash "$CHECKER" >"$FIXTURE_ROOT/usage.stdout" 2>"$FIXTURE_ROOT/usage.stderr"
USAGE_STATUS=$?
set -e
if [[ "$MISSING_STATUS" -ne 2 ]] \
  || ! grep -F -- "missing or unreadable LaTeX log" "$FIXTURE_ROOT/missing.stderr" >/dev/null \
  || [[ -s "$FIXTURE_ROOT/missing.stdout" ]]; then
  echo "formal PDF log self-test: missing-log control did not fail closed" >&2
  exit 1
fi
if [[ "$USAGE_STATUS" -ne 2 ]] \
  || ! grep -F -- "usage:" "$FIXTURE_ROOT/usage.stderr" >/dev/null \
  || [[ -s "$FIXTURE_ROOT/usage.stdout" ]]; then
  echo "formal PDF log self-test: usage control did not fail closed" >&2
  exit 1
fi

echo "OK: $REJECTED_COUNT rejected diagnostic cases; $ACCEPTED_COUNT clean/convergence and two interface controls pass"
