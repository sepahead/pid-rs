function Table(t)
  assert(#t.colspecs == 6, "unexpected table shape")
  local widths = {0.20,0.16,0.16,0.16,0.16,0.16}
  for i=1,6 do t.colspecs[i][2]=widths[i] end

end
function Link(l)
  if not l.target:match("^[%a][%w+.-]*:") and not l.target:match("^#") then
    l.target="../../"..l.target
  end
  return l
end
