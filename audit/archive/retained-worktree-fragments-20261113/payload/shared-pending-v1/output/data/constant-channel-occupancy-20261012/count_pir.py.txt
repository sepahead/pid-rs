#!/usr/bin/env python3
"""Complete-recording integer PIR counts; entropy unevaluated."""
from __future__ import annotations

import argparse
import csv
import datetime as dt
import hashlib
import io
import json
import os
from pathlib import Path
import re
import stat
import sys

RAW_CAP = 4 * 1024**2
OUTPUT_CAP = 16384
EXPECTED_BYTES = 931630
EXPECTED_ROWS = 10129
EXPECTED_SHA = "c090ee5c94b61762bfec9d22767864490b51030f98cca030f02822468db25d9c"
HEADER = (
    "Date", "Time", "S1_Temp", "S2_Temp", "S3_Temp", "S4_Temp",
    "S1_Light", "S2_Light", "S3_Light", "S4_Light", "S1_Sound",
    "S2_Sound", "S3_Sound", "S4_Sound", "S5_CO2", "S5_CO2_Slope",
    "S6_PIR", "S7_PIR", "Room_Occupancy_Count",
)
TUPLES = ((0, 0), (0, 1), (1, 0), (1, 1))
NUMBER = re.compile(r"[+-]?(?:[0-9]+(?:\.[0-9]*)?|\.[0-9]+)(?:[eE][+-]?[0-9]+)?\Z")
DATE = re.compile(r"[0-9]{4}/[0-9]{2}/[0-9]{2}\Z")
TIME = re.compile(r"[0-9]{2}:[0-9]{2}:[0-9]{2}\Z")


def need(condition, message):
    if not condition:
        raise RuntimeError(message)


def identity(info):
    return (info.st_dev, info.st_ino, info.st_mode, info.st_nlink,
            info.st_size, info.st_mtime_ns, info.st_ctime_ns)


def captured(path, cap):
    path = path.absolute()
    need(path == path.resolve(strict=True), "noncanonical/symlink input")
    before = path.lstat()
    need(stat.S_ISREG(before.st_mode) and before.st_nlink == 1
         and 0 < before.st_size <= cap, "input type/byte cap")
    descriptor = os.open(path, os.O_RDONLY | os.O_NOFOLLOW)
    try:
        opened = os.fstat(descriptor)
        with os.fdopen(descriptor, "rb", closefd=False) as stream:
            raw = stream.read(cap + 1)
        after = os.fstat(descriptor)
    finally:
        os.close(descriptor)
    need(identity(before) == identity(opened) == identity(after)
         == identity(path.lstat()) and len(raw) == before.st_size,
         "input changed during capture")
    return path, raw, identity(before)


def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def stamp(date, clock):
    need(DATE.fullmatch(date) and TIME.fullmatch(clock), "timestamp grammar")
    text = date + " " + clock
    value = dt.datetime.strptime(text, "%Y/%m/%d %H:%M:%S")
    return value, text


def analyse(raw):
    # Parse captured bytes; no float or logarithm evaluation.
    csv.field_size_limit(128)
    reader = csv.reader(io.StringIO(raw.decode("ascii", "strict"), newline=""), strict=True)
    need(tuple(next(reader, ())) == HEADER, "exact CSV header mismatch")
    joint = [[0] * 4 for _ in TUPLES]
    days = {}
    first = last = previous = None
    interval_min = interval_max = None
    total = 0
    for line, row in enumerate(reader, start=2):
        need(total < EXPECTED_ROWS and len(row) == len(HEADER), "row count/arity")
        need(all(0 < len(cell) <= 64 for cell in row), "missing/oversized cell")
        need(all(NUMBER.fullmatch(cell) for cell in row[2:16]),
             "nonfinite/invalid numeric spelling at row " + str(line))
        need(row[16] in {"0", "1"} and row[17] in {"0", "1"},
             "PIR alphabet at row " + str(line))
        need(row[18] in {"0", "1", "2", "3"}, "original target alphabet at row " + str(line))
        moment, text = stamp(row[0], row[1])
        if previous is not None:
            need(moment > previous, "non-increasing timestamp")
            difference = moment - previous
            seconds = difference.days * 86400 + difference.seconds
            interval_min = seconds if interval_min is None else min(interval_min, seconds)
            interval_max = seconds if interval_max is None else max(interval_max, seconds)
        first = text if first is None else first
        last, previous = text, moment
        key = row[0]
        if key not in days:
            need(len(days) < 16, "date-group resource cap")
            days[key] = {"date": key, "rows": 0, "first_timestamp": text,
                         "last_timestamp": text}
        days[key]["rows"] += 1
        days[key]["last_timestamp"] = text
        index = 2 * int(row[16]) + int(row[17])
        joint[index][int(row[18])] += 1
        total += 1
    need(total == EXPECTED_ROWS, "incomplete recording")
    counts = [sum(values) for values in joint]
    target_counts = [sum(row[t] for row in joint) for t in range(4)]
    need(sum(counts) == sum(target_counts) == sum(d["rows"] for d in days.values()) == total,
         "count conservation")
    departures = total - counts[0]
    zero_motion_occupied = sum(joint[0][1:])
    need(0 <= zero_motion_occupied <= counts[0] and 0 <= departures <= total,
         "count range")
    return {
        "N": total,
        "C": {"columns": ["S6_PIR", "S7_PIR"], "ordered_alphabet": [list(c) for c in TUPLES],
              "complete_alphabet_size": 4, "baseline_tuple": [0, 0], "counts": counts,
              "departures_from_baseline": departures,
              "departure_semantics": "not equipment fault"},
        "T": {"column": "Room_Occupancy_Count", "alphabet": [0, 1, 2, 3],
              "counts": target_counts, "transformation": "none"},
        "C_by_T": {"rows": [list(c) for c in TUPLES], "columns": [0, 1, 2, 3],
                   "integer_counts": joint},
        "zero_motion_nonzero_occupancy_count": zero_motion_occupied,
        "row_date_provenance": {
            "retention": "all rows, no selection or split",
            "timestamp_timezone": "unknown; naive strings",
            "first_timestamp": first, "last_timestamp": last,
            "strictly_increasing_timestamps": True,
            "min_interval_seconds": interval_min,
            "max_interval_seconds": interval_max,
            "dates": [days[key] for key in sorted(days)],
        },
        "empirical_entropy": {
            "status": "unevaluated_exact_formula",
            "expression": "-sum_(c:n_c>0) (n_c/N)*ln(n_c/N)",
            "definitions": {"n_c": "integer C count", "N": "all retained rows",
                            "ln": "natural logarithm; information unit nats"},
            "zero_convention": "0*ln(0)=0 by continuous extension",
            "terms": [{"C": list(c), "numerator": count, "denominator": total}
                      for c, count in zip(TUPLES, counts) if count > 0],
        },
        "scope": {"law": "complete empirical law", "remaining_X_encoding": "unbound",
                  "PID_population_utility_claims": "none"},

    }


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--input", required=True, type=Path)
    parser.add_argument("--output", required=True, type=Path)
    args = parser.parse_args()
    need(sys.flags.isolated and sys.flags.no_site and sys.dont_write_bytecode,
         "requires stock Python -I -S -B")
    path, raw, raw_identity = captured(args.input, RAW_CAP)
    need(len(raw) == EXPECTED_BYTES and digest(raw) == EXPECTED_SHA, "raw source pin mismatch")
    result = analyse(raw)
    result.update(schema="pid-rs-uci864-complete-pir-counts-v1",
                  status="integer_counts_only_entropy_unevaluated",
                  input={"path": str(path), "bytes": len(raw), "sha256": digest(raw),
                         "identity": list(raw_identity), "header": list(HEADER)})
    encoded = (json.dumps(result, sort_keys=True, indent=2, ensure_ascii=True,
                          allow_nan=False) + "\n").encode("ascii")
    need(len(encoded) <= OUTPUT_CAP, "output byte cap; no file opened")
    need(captured(path, RAW_CAP)[1:] == (raw, raw_identity), "input changed before output")
    output = args.output.absolute()
    need(output.parent == output.parent.resolve(strict=True) and output.parent.is_dir(),
         "noncanonical output parent")
    with output.open("xb") as stream:
        stream.write(encoded)
        stream.flush()
        os.fsync(stream.fileno())
    print("Integer counts only; entropy unevaluated; no PID/population claim.")
    return 0


if __name__ == "__main__":
    try:
        raise SystemExit(main())
    except (OSError, RuntimeError, ValueError, csv.Error, UnicodeError) as error:
        print("UCI864 counter refusal: " + str(error), file=sys.stderr)
        raise SystemExit(1)
