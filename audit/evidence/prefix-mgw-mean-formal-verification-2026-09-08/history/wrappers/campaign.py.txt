"""Proposed bounded MGW prefix mean candidate wrapper. Requires a later freeze and campaign."""

from __future__ import annotations

import argparse
import datetime
import fcntl
import hashlib

import os
import stat

import sys
from types import ModuleType


LANE = Path(__file__).resolve().parent
ROOT = LANE.parent.parent
PREP = LANE.parent / "sx-prefix-mgw-mean-type-preparation-20260906"
PASSED = PREP / "runs/full-normal-01"
RUNTIME_SHA = "bd8a9f2272a20422863c9902ce2148d2957471bb958d13949fece923cb6a7f5d"
PASSED_SHA = "edb2bd4bcb92c25276367a80b7a462c02cc8977163825f9314359e74450f0338"
DEPS = ['PidFiniteConvergence/Deterministic', 'PidFiniteConvergence/SxEventBridge', 'MgwBridgeDepsV1/SxDnf/Contract', 'MgwBridgeDepsV1/JoinLog/Contract', 'MgwBridgeDepsV1/SxDnf/JudgeCore', 'MgwBridgeDepsV1/SxDnf/Proofs', 'MgwBridgeDepsV1/JoinLog/Proofs', 'PidMgwBridge/Contract', 'PidMgwBridge/RawTargets', 'PidMgwBridge/AliasTargets', 'PidMgwBridge/JudgeCore', 'PidMgwBridge/Candidate', 'PidMgwBridge/SemanticJudge', 'PidPrefixProbability/Contract', 'PidPrefixProbability/RawTargets', 'PidPrefixProbability/AliasTargets', 'PidPrefixProbability/Candidate', 'PidPrefixProbability/SemanticJudge']
TRUSTED = ["PidPrefixMgwMean/" + name for name in ("Contract", "RawTargets", "AliasTargets")]
REQUIRED_WRAPPER_FILES = {
    "runtime.py", "policy.py", "prefix-policy.py", "mgw-policy.py", "ACCEPTED_DEPENDENCY_RECORDS.json", "campaign.py", "SemanticJudge.lean.in",
    "TASK.md", "SELECTION_RULE.json", "PREPARATION_INPUT_PINS.json", "SOURCE_GRAPH.json",
    "PREPARATION_ADOPTION.json", "OPERATIONAL_CONTROL_ACCEPTANCE.json", "PRIOR_CAMPAIGN_TERMINAL.json"} | {
        "sources/" + module + ".lean" for module in DEPS + TRUSTED}
PREPARATION_ADOPTION_SHA = "3a08e0c2fc46533bbd104858195d17339513fd1559d47cc9d4dea45c49583135"
# Definition-only adoption. Operational controls and a live freeze remain separate.
CONTRACT_SHA = "b5a2e6599e16f251cee6cfe13569eca459d1d5e4f276596bdc9e30e9be95b70a"
RAW_TARGET_SHA = "4461c2242b6c2de34b9bb968adf2da9182c10691dd5f33c0984f1356618a8bdc"



def utc():
    return datetime.datetime.now(datetime.timezone.utc)


def load_module(name, path, expected):
    raw = path.read_bytes()
    if hashlib.sha256(raw).hexdigest() != expected:
        raise RuntimeError("module bytes differ: " + str(path))
    module = ModuleType(name)
    module.__file__ = str(path)
    exec(compile(raw, str(path), "exec", dont_inherit=True, optimize=sys.flags.optimize), module.__dict__)
    return module


def main():
    if not (sys.flags.isolated and sys.flags.no_site and sys.dont_write_bytecode
            and sys.flags.optimize in (0, 1)):
        raise SystemExit("requires Python -I -S -B, optionally -O")
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("kind", choices=("development", "full", "wrong-last-target"))
    parser.add_argument("--candidate-sha256", required=True)
    args = parser.parse_args()
    j = load_module("prefix_mgw_mean_frozen_runtime", LANE / "runtime.py", RUNTIME_SHA)
    lock = (LANE / "EXECUTION.lock").open("a+")
    fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
    snapshot = j.Snapshot()
    freeze_raw = snapshot.read(LANE / "FROZEN.json")
    freeze = j.strict_json(freeze_raw)
    task_raw = snapshot.read(LANE / "CAMPAIGN.json")
    task = j.strict_json(task_raw)
    j.require(freeze.get("schema") == "pid-rs/prefix-mgw-mean-proof-judge-freeze-v2"
              and freeze.get("status") == "adopted_after_actual_controls_before_candidate"
              and freeze.get("candidate_exists") is False,
              "mean judge is not an adopted pre-candidate freeze")
    j.require(task.get("schema") == "pid-rs/prefix-mgw-mean-proof-campaign-v2",
              "campaign schema differs")
    j.require(task["freeze_sha256"] == j.digest(freeze_raw), "campaign freeze differs")
    for field, expected in (("max_development_attempts", 79), ("max_full_attempts", 8),
                            ("max_concurrent_wrappers", 1)):
        j.require(type(task.get(field)) is int and task[field] == expected,
                  "fixed future campaign bound differs: " + field)
    j.require(task.get("candidate_write_scope") == "candidate/Candidate.lean",
              "candidate write scope differs")
    registered = datetime.datetime.fromisoformat(task["registered_utc"])
    declared_deadline = datetime.datetime.fromisoformat(task["deadline_utc"])
    j.require(registered.tzinfo is not None and declared_deadline.tzinfo is not None
              and declared_deadline - registered == datetime.timedelta(minutes=720)
              and utc() >= registered, "campaign clock is not the registered 720-minute interval")
    j.require(type(freeze.get("wrapper_sha256")) is dict
              and REQUIRED_WRAPPER_FILES <= set(freeze["wrapper_sha256"]),
              "required wrapper/source/adoption pins absent")
    j.require(not (LANE / "STOP_RECORD.json").exists(), "campaign is stopped")
    deadline = datetime.datetime.fromisoformat(task["deadline_utc"])
    j.require(utc() < deadline, "campaign deadline exhausted")
    for name, sha in freeze["wrapper_sha256"].items():
        j.require(j.digest(snapshot.read(LANE / name)) == sha, "wrapper drift: " + name)
    expected_preparation = j.strict_json(snapshot.read(LANE / "PREPARATION_INPUT_PINS.json"))
    j.require(freeze["preparation_sha256"] == expected_preparation,
              "preparation pin roster differs from reviewed source proposal")
    for name, sha in expected_preparation.items():
        j.require(j.digest(snapshot.read(PREP / name)) == sha, "preparation drift: " + name)
    adoption_raw = snapshot.read(LANE / "PREPARATION_ADOPTION.json")
    adoption = j.strict_json(adoption_raw)
    j.require(j.digest(adoption_raw) == PREPARATION_ADOPTION_SHA
              and adoption.get("status") == "three_exact_mean_target_definitions_adopted_for_prospective_candidate_preparation"
              and adoption.get("contract_sha256") == CONTRACT_SHA
              and adoption.get("raw_targets_sha256") == RAW_TARGET_SHA
              and type(adoption.get("new_mean_theorems")) is int
              and adoption["new_mean_theorems"] == 0,
              "completed preparation lacks exact root adoption")
    prior_raw = snapshot.read(LANE / "PRIOR_CAMPAIGN_TERMINAL.json")
    prior = j.strict_json(prior_raw)
    j.require(j.digest(prior_raw) == "cae0627801b6375082b8619c3d404a041e6f6f4d6321531c50c40d9444b8fead"
              and prior.get("prior_status") == "closed_incomplete_original_deadline_exhausted"
              and type(prior.get("actual_development_attempts")) is int
              and prior["actual_development_attempts"] == 1
              and type(prior.get("actual_full_attempts")) is int
              and prior["actual_full_attempts"] == 0
              and type(prior.get("accepted_mean_theorems")) is int
              and prior["accepted_mean_theorems"] == 0,
              "prior terminal campaign boundary differs")
    for name, record in prior["files"].items():
        historical_raw = snapshot.read(LANE / "previous-terminal" / name)
        j.require(len(historical_raw) == record["bytes"]
                  and j.digest(historical_raw) == record["sha256"],
                  "prior terminal preimage differs: " + name)
    controls = j.strict_json(snapshot.read(LANE / "OPERATIONAL_CONTROL_ACCEPTANCE.json"))
    j.require(controls.get("status") == "adopted_completed_operational_controls"
              and type(controls.get("new_mean_theorems_proved")) is int
              and controls["new_mean_theorems_proved"] == 0,
              "wrapper controls lack completed adoption")
    j.require(task.get("selection_rule_sha256") == j.digest(snapshot.read(LANE / "SELECTION_RULE.json")),
              "selection rule differs")
    policy = load_module("prefix_mgw_mean_frozen_policy", LANE / "policy.py", freeze["wrapper_sha256"]["policy.py"])
    prefix_policy = load_module("accepted_prefix_dependency_policy", LANE / "prefix-policy.py",
                                freeze["wrapper_sha256"]["prefix-policy.py"])
    mgw_policy = load_module("accepted_mgw_dependency_policy", LANE / "mgw-policy.py",
                             freeze["wrapper_sha256"]["mgw-policy.py"])
    accepted_records = j.strict_json(snapshot.read(LANE / "ACCEPTED_DEPENDENCY_RECORDS.json"))
    preparation_receipt = j.strict_json(snapshot.read(PASSED / "receipt.json"))
    j.require(j.digest(snapshot.read(PASSED / "receipt.json")) == PASSED_SHA,
              "mean type preparation receipt differs")
    j.require(set(accepted_records) == {"accepted_prefix_dependency", "accepted_mgw_dependency"},
              "accepted proof family roster differs")
    for family, records in accepted_records.items():
        j.require(records == preparation_receipt["semantic_records"][family],
                  "accepted dependency records differ from reviewed mean preparation")
    expected_prefix = accepted_records["accepted_prefix_dependency"]
    expected_mgw = accepted_records["accepted_mgw_dependency"]
    for name, expected in j.strict_json(snapshot.read(PREP / "NEW_IMPORT_SOURCE_PINS.json")).items():
        j.require(j.digest(snapshot.read(ROOT / name)) == expected, "explicit import source drift: " + name)
    j.verify_source_pins(ROOT, snapshot)
    helper, base = j.load_helpers(ROOT, snapshot)
    base.check_toolchain()
    base.check_lakefile()
    base.check_manifest()
    git = base.find_git()
    j.require(Path(base.run_git(git, ROOT, ["rev-parse", "--show-toplevel"], "root check").strip()) == ROOT,
              "repository root differs")
    base.check_dependency_checkouts(git)
    candidate = LANE / "candidate/Candidate.lean"
    ledger_path = LANE / "ATTEMPT_LEDGER.json"
    ledger = j.strict_json(ledger_path.read_bytes()) if ledger_path.exists() else {
        "campaign_sha256": j.digest(task_raw), "attempts": []}
    j.require(ledger["campaign_sha256"] == j.digest(task_raw), "campaign ledger identity differs")
    slot_kind = "development" if args.kind == "development" else "full"
    used = sum(item["slot_kind"] == slot_kind for item in ledger["attempts"])
    j.require(used < task["max_" + slot_kind + "_attempts"], "attempt limit exhausted")
    attempts = LANE / "attempts"
    attempts.mkdir(exist_ok=True)
    output = attempts / (str(len(ledger["attempts"]) + 1).zfill(3) + "-" + args.kind
                         + "-" + str(sys.flags.optimize))
    output.mkdir(exist_ok=False)
    entry = {"registered_utc": utc().isoformat(), "kind": args.kind, "slot_kind": slot_kind,
             "mode": sys.flags.optimize, "candidate_sha256": args.candidate_sha256,
             "output": str(output.relative_to(LANE)),
             "prospective_compiler_kernel_children": 1 if args.kind == "development" else
                 (23 if args.kind == "wrong-last-target" else 25)}
    (output / "REGISTERED.json").write_bytes(j.canonical(entry))
    ledger["attempts"].append(entry)
    temporary = LANE / "ATTEMPT_LEDGER.next.json"
    with temporary.open("xb") as stream:
        stream.write(j.canonical(ledger))
        stream.flush()
        os.fsync(stream.fileno())
    temporary.replace(ledger_path)
    result = {"status": "failed", "registered": entry}
    runner = None
    try:
        source_stat = candidate.lstat()
        result["observed_candidate_metadata"] = {
            "size": source_stat.st_size, "mode": source_stat.st_mode,
            "mtime_ns": source_stat.st_mtime_ns}
        j.require(stat.S_ISREG(source_stat.st_mode), "candidate must be a regular file")
        j.require(source_stat.st_size <= 2 * 1024**2, "candidate exceeds pre-read byte budget")
        source = snapshot.read(candidate)
        result["captured_sha256"] = j.digest(source)
        (output / "Candidate.original.lean").write_bytes(source)
        j.check_candidate_hash(source, args.candidate_sha256)
        policy.candidate_policy(source, base, candidate,
                                development_helpers=args.kind == "development")
        if args.kind == "wrong-last-target":
            prior = []
            for attempt in ledger["attempts"][:-1]:
                if attempt["kind"] == "full" and attempt["candidate_sha256"] == j.digest(source):
                    directory = LANE / attempt["output"]
                    prior_path = directory / "RESULT.json"
                    if prior_path.exists():
                        prior_raw = snapshot.read(prior_path)
                        body = j.strict_json(prior_raw)
                        original = snapshot.read(directory / "Candidate.original.lean")
                        if (body["status"] == "fresh_semantic_kernel_pass"
                                and body["registered"] == attempt and original == source):
                            prior.append({"path": str(prior_path), "sha256": j.digest(prior_raw)})
            j.require(prior, "negative control requires a prior full pass of identical original bytes")
            result["prior_full_passes"] = prior
            marker = "theorem prefix_expectations_to_mgw_mean"
            text = source.decode()
            masked = base.mask_lean_comments_and_strings(text, candidate)
            j.require(masked.count(marker) == 1, "negative-control final theorem marker differs")
            source = (text[:masked.index(marker)]
                      + "theorem prefix_expectations_to_mgw_mean : True := by trivial\n\nend PidPrefixMgwMeanCandidate\n").encode()
            policy.candidate_policy(source, base, output / "Candidate.control.lean", final_control=True)
        (output / "Candidate.executed.lean").write_bytes(source)
        result["executed_sha256"] = j.digest(source)
        (output / "EXECUTION_INPUT.json").write_bytes(j.canonical({
            "registered": entry, "executed_sha256": j.digest(source)}))
        old_raw = snapshot.read(PASSED / "receipt.json")
        j.require(j.digest(old_raw) == PASSED_SHA, "passed preparation receipt differs")
        old = j.strict_json(old_raw)
        env_raw = snapshot.read(PASSED / "environment.json")
        j.require(j.digest(env_raw) == old["artifacts"]["environment.json"], "tool environment differs")
        observed = j.strict_json(env_raw)
        tools = observed["tools"]
        for name, sha in tools.items():
            j.require(j.digest(snapshot.read(Path(name))) == sha, "tool bytes differ")
        lean = next(Path(name) for name in tools if name.endswith("/bin/lean"))
        checker = next(Path(name) for name in tools if name.endswith("/bin/leanchecker"))
        src, build = output / "src", output / "build"
        src.mkdir()
        build.mkdir()
        for module in DEPS + TRUSTED:
            name = "sources/" + module + ".lean"
            raw = snapshot.read(LANE / name)
            j.require(j.digest(raw) == freeze["wrapper_sha256"][name]
                      == freeze["preparation_sha256"][name], "module pin absent or changed")
            target = src / (module + ".lean")
            target.parent.mkdir(parents=True, exist_ok=True)
            target.write_bytes(raw)
            snapshot.read(target)
            if args.kind == "development":
                artifact_name = "build/" + module + ".olean"
                raw_object = snapshot.read(PASSED / artifact_name)
                j.require(j.digest(raw_object) == old["artifacts"][artifact_name], "development object changed")
                target_object = build / (module + ".olean")
                target_object.parent.mkdir(parents=True, exist_ok=True)
                target_object.write_bytes(raw_object)
                snapshot.read(target_object)
        candidate_copy = src / "PidPrefixMgwMean/Candidate.lean"
        candidate_copy.write_bytes(source)
        snapshot.read(candidate_copy)
        semantic = snapshot.read(LANE / "SemanticJudge.lean.in")
        (src / "PidPrefixMgwMean/SemanticJudge.lean").write_bytes(semantic)
        snapshot.read(src / "PidPrefixMgwMean/SemanticJudge.lean")
        environment = {"PATH": str(lean.parent) + ":/usr/bin:/bin", "LANG": "C.UTF-8", "LC_ALL": "C.UTF-8",
                       "LEAN_PATH": str(build) + os.pathsep + os.pathsep.join(observed["lean_search_path"].split(os.pathsep)[1:]),
                       "LEAN_SRC_PATH": str(src)}
        runner = j.Runner(output, environment, preparation=False)
        modules = ([] if args.kind == "development" else DEPS + TRUSTED)
        modules += ["PidPrefixMgwMean/Candidate"]
        if args.kind != "development":
            modules += ["PidPrefixMgwMean/SemanticJudge"]
        for module in modules:
            snapshot.verify()
            target = build / (module + ".olean")
            target.parent.mkdir(parents=True, exist_ok=True)
            j.require(not (LANE / "STOP_RECORD.json").exists(), "campaign is stopped before child launch")
            remaining = (deadline - utc()).total_seconds()
            j.require(remaining > 0, "campaign deadline exhausted before child launch")
            try:
                stdout = runner.run([str(lean), "-t", "0", "-M", "4096", "-o", str(target),
                                     str(src / (module + ".lean"))], src,
                                    "compile-" + module.replace("/", "-"), timeout=min(600, remaining),
                                    allow_stdout=(module.endswith("SemanticJudge") or args.kind == "development"))
            except j.JudgeError:
                if args.kind == "wrong-last-target" and module == "PidPrefixMgwMean/SemanticJudge":
                    last = runner.commands[-1]
                    log = output / (str(len(runner.commands)-1).zfill(2) + "-" + last["label"]) / "stdout.log"
                    text = log.read_text()
                    j.require(last["failure"] is None and last["returncode"] != 0
                              and "prefix_expectations_to_mgw_mean" in text and "type mismatch" in text.lower(),
                              "negative control did not reach its target-type mismatch")
                    result["status"] = "expected_target_rejection"
                    break
                raise
            snapshot.read(target)
            if module == "PidPrefixMgwMean/SemanticJudge":
                result["semantic_records"] = policy.semantic_records(stdout, j.strict_json)
            elif module == "PidPrefixProbability/SemanticJudge":
                parsed_prefix = prefix_policy.semantic_records(stdout, j.strict_json)
                j.require(parsed_prefix == expected_prefix, "accepted six-proof dependency records differ")
                result["accepted_prefix_dependency_records"] = parsed_prefix
            elif module == "PidMgwBridge/SemanticJudge":
                parsed_mgw = mgw_policy.semantic_records(stdout, j.strict_json)
                j.require(parsed_mgw == expected_mgw, "accepted eleven-proof dependency records differ")
                result["accepted_mgw_dependency_records"] = parsed_mgw
        else:
            j.require(args.kind != "wrong-last-target", "incorrect target was accepted")
            if args.kind == "full":
                root_source = src / "PrefixMgwMeanCandidateRoot.lean"
                root_source.write_text("import PidMgwBridge.SemanticJudge\nimport PidPrefixProbability.SemanticJudge\nimport PidPrefixMgwMean.SemanticJudge\n")
                snapshot.read(root_source)
                for label, command in (
                    ("compile-root", [str(lean), "-t", "0", "-M", "4096", "-o", str(build / "PrefixMgwMeanCandidateRoot.olean"), str(root_source)]),
                    ("fresh-kernel", [str(checker), "--fresh", "PrefixMgwMeanCandidateRoot"]),
                ):
                    snapshot.verify()
                    j.require(not (LANE / "STOP_RECORD.json").exists(), "campaign is stopped before child launch")
                    remaining = (deadline - utc()).total_seconds()
                    j.require(remaining > 0, "campaign deadline exhausted before child launch")
                    runner.run(command, src, label, timeout=min(600, remaining))
                    if label == "compile-root":
                        snapshot.read(build / "PrefixMgwMeanCandidateRoot.olean")
            result["status"] = "fresh_semantic_kernel_pass" if args.kind == "full" else "development_compile_pass"
        snapshot.verify()
        j.require(not (LANE / "STOP_RECORD.json").exists(), "campaign is stopped before custody closure")
        j.require(utc() < deadline, "campaign ended before custody closure")
    except BaseException as error:
        result.update(status="failed", failure=str(error))
    finally:
        result["commands"] = [] if runner is None else runner.commands
        result["inputs"] = snapshot.manifest()
        try:
            snapshot.save(output)
        except BaseException as error:
            result.update(status="failed", preservation_failure=str(error))
        artifacts = {}
        try:
            for path in sorted(output.rglob("*")):
                mode = path.lstat().st_mode
                if stat.S_ISDIR(mode):
                    continue
                j.require(stat.S_ISREG(mode), "nonregular retained artifact")
                raw = path.read_bytes()
                artifacts[str(path.relative_to(output))] = {"bytes": len(raw), "sha256": j.digest(raw)}
        except BaseException as error:
            result.update(status="failed", artifact_failure=str(error))
        result["artifacts"] = artifacts
        try:
            snapshot.verify()
            j.require(not (LANE / "STOP_RECORD.json").exists(),
                      "campaign stopped during final evidence preservation")
            completed = utc()
            j.require(completed < deadline, "campaign expired during final evidence preservation")
        except BaseException as error:
            result.update(status="failed", final_custody_failure=str(error))
        result["completed_utc"] = utc().isoformat()
        (output / "RESULT.json").write_bytes(j.canonical(result))
        print(j.canonical({"status": result["status"], "output": str(output)}).decode(), end="")
    return 1 if result["status"] == "failed" else 0


if __name__ == "__main__":
    raise SystemExit(main())
