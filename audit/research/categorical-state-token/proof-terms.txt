theorem D1Token.read_iff : ∀ (word t rest : List UInt8),
  D1Token.read word = some (t, rest) ↔ Admissible t ∧ word = emit t ++ rest :=
fun word t rest =>
  { mp := read_sound word t rest, mpr := fun a => And.casesOn a fun ht right => Eq.symm right ▸ read_emit t rest ht }
theorem D1Token.unique : ∀ (word a b sa sb : List UInt8),
  D1Token.read word = some (a, sa) → D1Token.read word = some (b, sb) → a = b ∧ sa = sb :=
fun word a b sa sb ha hb =>
  Eq.mp (Eq.trans (Option.some.injEq (a, sa) (b, sb)) (Prod.mk.injEq a sa b sb))
    (Eq.mp (congrArg (fun _a => _a = some (b, sb)) ha) hb)
