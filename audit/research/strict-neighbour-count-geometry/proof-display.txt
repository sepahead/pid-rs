theorem W7.attainable_iff : ∀ (n k x y : Nat),
  0 < k → k < n → ((∃ rows r, rows.length + 1 = n ∧ Realizes rows r k x y) ↔ Triangle n k x y) :=
fun n k x y hk hn =>
  {
    mp := fun a =>
      Exists.casesOn a fun rows h =>
        Exists.casesOn h fun r h =>
          And.casesOn h fun hlen obs =>
            Eq.mpr (id (congrArg (fun _a => Triangle _a k x y) (Eq.symm hlen))) (necessary rows r k x y obs),
    mpr := fun a =>
      And.casesOn a fun hkx right =>
        And.casesOn right fun hxn right =>
          And.casesOn right fun hky right =>
            And.casesOn right fun hyn hs =>
              have ht := Decidable.byContradiction fun a => attainable_iff._proof_1_1 n k x y hn hkx hky hs a;
              And.casesOn (constructed (k - 1) (x - k) (y - k) (n + k - x - y) ht) fun hlen hobs =>
                have hK := Decidable.byContradiction fun a => attainable_iff._proof_1_2 n k x y hk a;
                have hX := Decidable.byContradiction fun a => attainable_iff._proof_1_3 n k x y hk hkx hky hs hlen hK a;
                have hY := Decidable.byContradiction fun a => attainable_iff._proof_1_4 n k x y hk hky hs hlen hK hX a;
                Exists.intro (witness (k - 1) (x - k) (y - k) (n + k - x - y))
                  (Exists.intro 2
                    ⟨Decidable.byContradiction fun a => attainable_iff._proof_1_5 n k x y hk hs hlen hK hX hY a,
                      Eq.mp
                        (congrArg (fun _a => Realizes (witness (k - 1) (x - k) (y - k) (n + k - x - y)) 2 _a x y) hK)
                        (Eq.mp
                          (congrArg
                            (fun _a => Realizes (witness (k - 1) (x - k) (y - k) (n + k - x - y)) 2 (k - 1 + 1) x _a)
                            hY)
                          (Eq.mp
                            (congrArg
                              (fun _a =>
                                Realizes (witness (k - 1) (x - k) (y - k) (n + k - x - y)) 2 (k - 1 + 1) _a
                                  (k - 1 + (y - k) + 1))
                              hX)
                            hobs))⟩) }
